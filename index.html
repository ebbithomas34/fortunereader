// AUTO-LOAD FROM GITHUB async function autoLoadGitHub() { const urls = [ 'https://raw.githubusercontent.com/ebbithomas34/fortunereader/main/data.csv', './data.csv', 'https://cdn.jsdelivr.net/gh/ebbithomas34/fortunereader@main/data.csv' ]; showMessage('Loading assessment data...', 'info'); for (const url of urls) { try { console.log('Trying to load from:', url); const response = await fetch(url); if (response.ok) { const csvText = await response.text(); console.log('Successfully loaded from:', url); console.log('CSV length:', csvText.length); Papa.parse(csvText, { header: true, skipEmptyLines: true, complete: function(results) { console.log('Parsed rows:', results.data.length); parseData(results.data); }, error: function(error) { console.error('Parse error:', error); showMessage('Error parsing data: ' + error.message, 'error'); } }); return; // Success - exit function } else { console.log('Failed to load from:', url, 'Status:', response.status); } } catch (error) { console.log('Error loading from:', url, error.message); continue; // Try next URL } } // All URLs failed - show manual upload option console.error('All data sources failed'); showMessage('Could not load data automatically. Click "Begin Assessment" to upload data.csv', 'info'); document.getElementById('beginBtn').style.display = 'block'; }
